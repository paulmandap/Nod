    mask=0x3f,

    fileType=0x1,
    subtype=0x0,
    date=(0, 0)
  ),
  kids=[
    StringFileInfo([
      StringTable(

         StringStruct('FileDescription', 'Nod - a voice assistant for your desktop'),

         StringStruct('ProductName', 'Nod'),
         StringStruct('ProductVersion', '0.1.0')])
    ]),
    VarFileInfo([VarStruct('Translation', [1033, 1200])])
  ]
)
